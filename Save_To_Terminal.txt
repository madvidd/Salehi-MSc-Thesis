set +e
set +u
set +o pipefail 2>/dev/null

BASE=/home/server00/M
POINTER="$BASE/Results/LATEST_SHARP_AV2_TEMPORAL_AGENT_MAMBA_ROTATIONFIX80_RUN.txt"
DEST="$BASE/Terminal/Terminal.txt"

mkdir -p "$(dirname "$DEST")"

if [ ! -s "$POINTER" ]; then
  echo "ERROR: run pointer not found:"
  echo "$POINTER"
else
  RUN=$(tr -d '\r\n' < "$POINTER")

  {
    echo "Lab 2 corrected temporal-agent Mamba terminal snapshot"
    echo "Saved: $(date)"
    echo "Run: $RUN"
    echo

    if [ -f "$RUN/RUN_SETTINGS.txt" ]; then
      echo "===== RUN SETTINGS ====="
      cat "$RUN/RUN_SETTINGS.txt"
      echo
    fi

    if [ -f "$RUN/preflight/preflight.log" ]; then
      echo "===== FOUR-GPU PREFLIGHT ====="
      tr '\r' '\n' < "$RUN/preflight/preflight.log"
      echo
    fi

    if [ -f "$RUN/preflight/PREFLIGHT_PASSED.txt" ]; then
      echo "===== PREFLIGHT STATUS ====="
      cat "$RUN/preflight/PREFLIGHT_PASSED.txt"
      echo
    fi

    if [ -f "$RUN/full_run.log" ]; then
      echo "===== FULL FOUR-GPU TRAINING ====="
      tr '\r' '\n' < "$RUN/full_run.log"
      echo
    fi

    echo "===== SAVED CHECKPOINTS ====="
    find "$RUN" -name "*.ckpt" \
      -printf "%TY-%Tm-%Td %TH:%TM  %s bytes  %p\n" |
      sort
  } > "$DEST"

  sync

  echo "Saved one-time Lab 2 snapshot:"
  echo "$DEST"
  ls -lh "$DEST"
fi

echo "Terminal remains open."









Lab 3:

set +e
set +u
set +o pipefail 2>/dev/null

BASE=/home/server01/M
RESULTS_POINTER="$BASE/Results/LATEST_SHARP_ATTENTION_ABLATION.txt"
DEST="$BASE/Terminal/Terminal.txt"

mkdir -p "$(dirname "$DEST")"

if [ ! -s "$RESULTS_POINTER" ]; then
  echo "ERROR: results pointer not found:"
  echo "$RESULTS_POINTER"
else
  RESULTS=$(tr -d '\r\n' < "$RESULTS_POINTER")

  SUITE_LOG=$(find "$RESULTS" -maxdepth 1 \
    -type f -name "remaining_attention_suite_*.log" \
    -printf "%T@ %p\n" 2>/dev/null |
    sort -nr |
    head -1 |
    cut -d' ' -f2-)

  {
    echo "Lab 3 remaining-attention terminal snapshot"
    echo "Saved: $(date)"
    echo "Results root: $RESULTS"
    echo "Suite log: $SUITE_LOG"
    echo

    if [ -n "$SUITE_LOG" ] && [ -f "$SUITE_LOG" ]; then
      echo "===== REMAINING ATTENTION SUITE ====="
      tr '\r' '\n' < "$SUITE_LOG"
      echo
    fi

    for VARIANT in qknorm talking_heads qknorm_talking_heads; do
      echo "===== $VARIANT ====="

      if [ -f "$RESULTS/$VARIANT/train.log" ]; then
        echo "----- TRAINING LOG -----"
        tr '\r' '\n' < "$RESULTS/$VARIANT/train.log"
      fi

      if [ -f "$RESULTS/$VARIANT/eval.log" ]; then
        echo "----- EVALUATION LOG -----"
        tr '\r' '\n' < "$RESULTS/$VARIANT/eval.log"
      fi

      if [ -f "$RESULTS/$VARIANT/metrics.json" ]; then
        echo "----- METRICS -----"
        cat "$RESULTS/$VARIANT/metrics.json"
      fi

      echo
    done

    echo "===== PRESERVED PARTIAL RUNS ====="
    find "$RESULTS/preserved_partial_runs" -type f \
      -printf "%TY-%Tm-%Td %TH:%TM  %s bytes  %p\n" \
      2>/dev/null | sort
    echo

    echo "===== CHECKPOINTS ====="
    find "$RESULTS" -name "*.ckpt" \
      -printf "%TY-%Tm-%Td %TH:%TM  %s bytes  %p\n" \
      2>/dev/null | sort

  } > "$DEST"

  sync

  echo "Saved one-time Lab 3 snapshot:"
  echo "$DEST"
  ls -lh "$DEST"
fi

echo
echo "Terminal remains open."

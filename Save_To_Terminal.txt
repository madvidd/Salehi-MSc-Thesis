Lab 2:

set +e
set +u
set +o pipefail 2>/dev/null

BASE=/home/server00/M
POINTER="$BASE/Results/LATEST_SHARP_AV2_TEMPORAL_AGENT_MAMBA_ROTATIONFIX80_RUN.txt"
DEST="$BASE/Terminal/Terminal.txt"

mkdir -p "$(dirname "$DEST")"

if [ ! -s "$POINTER" ]; then
  echo "ERROR: run pointer not found:"
  echo "$POINTER"
else
  RUN=$(tr -d '\r\n' < "$POINTER")

  {
    echo "Lab 2 temporal-agent Mamba terminal snapshot"
    echo "Saved: $(date)"
    echo "Run: $RUN"
    echo

    if [ -f "$RUN/RUN_SETTINGS.txt" ]; then
      echo "===== RUN SETTINGS ====="
      cat "$RUN/RUN_SETTINGS.txt"
      echo
    fi

    if [ -f "$RUN/preflight/preflight.log" ]; then
      echo "===== FOUR-GPU PREFLIGHT ====="
      tr '\r' '\n' < "$RUN/preflight/preflight.log"
      echo
    fi

    if [ -f "$RUN/full_run.log" ]; then
      echo "===== FULL TRAINING ====="
      tr '\r' '\n' < "$RUN/full_run.log"
      echo
    fi

    echo "===== CHECKPOINTS ====="
    find "$RUN" -name "*.ckpt" \
      -printf "%TY-%Tm-%Td %TH:%TM  %s bytes  %p\n" |
      sort
  } > "$DEST"

  sync

  echo "Saved one-time snapshot:"
  echo "$DEST"
  ls -lh "$DEST"
fi

echo
echo "Terminal remains open."










Lab 3:
set +e
set +u
set +o pipefail 2>/dev/null

BASE=/home/server01/M
RESULTS=$(tr -d '\r\n' < \
  "$BASE/Results/LATEST_SHARP_ATTENTION_ABLATION.txt")

DEST="$BASE/Terminal/Terminal.txt"
mkdir -p "$(dirname "$DEST")"

SOURCE=$(find "$RESULTS" -maxdepth 1 -type f \
  -name "remaining_attention_suite_*.log" \
  -printf "%T@ %p\n" 2>/dev/null |
  sort -nr |
  head -1 |
  cut -d" " -f2-)

if [ -z "$SOURCE" ]; then
  SOURCE=$(find "$RESULTS" -type f -name "train.log" \
    -printf "%T@ %p\n" 2>/dev/null |
    sort -nr |
    head -1 |
    cut -d" " -f2-)
fi

if [ -n "$SOURCE" ] && [ -f "$SOURCE" ]; then
  tr '\r' '\n' < "$SOURCE" > "$DEST"
  sync

  echo "Lab 3 terminal snapshot saved:"
  echo "Source: $SOURCE"
  echo "Output: $DEST"
  ls -lh "$DEST"
else
  echo "ERROR: no current attention-suite log was found."
fi

echo "Terminal remains open."
